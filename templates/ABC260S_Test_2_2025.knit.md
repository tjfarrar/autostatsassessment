\item \textit{[11 marks]} Below are given the final marks of a class of fifteen students.

\begin{table}[h]
 {\huge\begin{tabular}{ccccccccccccccc}
  40 & 57 & 31 & 84 & 37 & 58 & 43 & 54 & 71 & 40 & 83 & 61 & 70 & 72 & 29\\
 \end{tabular}}
\end{table}

\beq

  \item \textit{[3 marks]} Find the mean of the final marks.

\ifmemo
\textcolor{red}{\begin{align*}
  \bar{x} &= \dfrac{1}{n}\displaystyle\sum_{i=1}^{n} x_i \text{ \tick} \\
  &= \dfrac{1}{15}\left[40 + 57 + 31 + 84 + 37 + 58 + 43 + 54 + 71 + 40 + 83 + 61 + 70 + 72 + 29\right] \text{ \tick} \\
  &= \dfrac{830}{15} = 55.3333 \text{ \tick}
\end{align*}}
\fi


  \item \textit{[3 marks]} Find the median of the final marks.  



\ifmemo
 \textcolor{red}{First we put the data in increasing order:
 \begin{table}[h]
 \begin{tabular}{ccccccccccccccc}
  29 & 31 & 37 & 40 & 40 & 43 & 54 & 57 & 58 & 61 & 70 & 71 & 72 & 83 & 84\\
 \end{tabular} 
\end{table}
 \tick \newline
 Then we take the 8th value. \tick. Hence, the median is 57 \tick}
\fi

  \item \textit{[3 marks]} Find the standard deviation of the final marks.  

\ifmemo
\textcolor{red}{\begin{align*}
  s^2 &= \dfrac{1}{n-1}\displaystyle\sum_{i=1}^{n} (x_i-\bar{x})^2 \text{ \tick} \\
  &= \dfrac{1}{14}\left[\left(40-\frac{830}{15}\right)^2 + \left(57-\frac{830}{15}\right)^2 + \cdots + \left(29-\frac{830}{15}\right)^2\right] \text{ \tick} \\
  s &= \sqrt{329.5238095} \\
  &= 18.1528 \text{ \tick}
\end{align*}}

\item \textit{[9 marks]} Consider the random variable $X$ with probability density function as follows:

  \[ f_X(x) = \begin{dcases*} k(6 x + 6 x^{2})  & for $0\le x \le 3$ \\ 0 & otherwise  \end{dcases*} \]
  
\beq
  \item \textit{[3 marks]} Find the value of $k$.

\textcolor{red}{\begin{align*} k\displaystyle\int_{0}^{3} f_X(x) \mathrm{d}x &= 1 \text{ \tick} \\ k\displaystyle\int_{0}^{3}6 x + 6 x^{2}\mathrm{d}x &= 1 \\
    k\left[3 x^{2} + 2 x^{3}\right]_{0}^{3} &= 1 \text{ \tick}\\ k\left(81\right)&= 1 \\ k= \frac{1}{81}&= 0.012346 \text{ \tick} \end{align*}}





   \item \textit{[3 marks]} Find \(\mathrm{Pr}(X < 1)\).

\textcolor{red}{\begin{align*}
      \mathrm{Pr}\left(X < 1\right) &= \displaystyle\int_{0}^{1}6 x + 6 x^{2}\mathrm{d}x \text{ \tick} \\ &= \left[3 x^{2} + 2 x^{3}\right]_{0}^{1}  \text{ \tick}\\ &= \left(\dfrac{5}{81}\right) - \left(0\right)  \\ &= \dfrac{5}{81} = 0.06173 \text{ \tick} \end{align*}}
